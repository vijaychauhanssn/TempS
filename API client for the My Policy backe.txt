const BASE_URL = `${process.env.EXPO_PUBLIC_BACKEND_URL}/api`;

async function request<T>(path: string, init?: RequestInit): Promise<T> {
  const res = await fetch(`${BASE_URL}${path}`, {
    ...init,
    headers: { "Content-Type": "application/json", ...(init?.headers ?? {}) },
  });
  if (!res.ok) {
    let message = "Something went wrong. Please try again.";
    try {
      const data = await res.json();
      if (typeof data.detail === "string") message = data.detail;
    } catch {
      // keep default message
    }
    throw new Error(message);
  }
  return res.json() as Promise<T>;
}

export const api = {
  listPolicies: () => request<Policy[]>("/policies"),
  getPolicy: (id: string) => request<Policy>(`/policies/${id}`),
  addPolicy: (body: AddPolicyInput) =>
    request<Policy>("/policies", { method: "POST", body: JSON.stringify(body) }),
  getInsurers: () => request<Insurer[]>("/insurers"),
};

// Native storage (Metro auto-picks index.web.ts on web — do NOT add Platform.OS checks).
//
// Import the ready-made singleton BY NAME and call methods on it — never a default
// import, never the methods bare:
//   import { storage } from "@/src/utils/storage";
//   await storage.getItem(key, fallback);      // the `fallback` arg is REQUIRED
//
// Namespaces: general KV -> getItem/setItem/removeItem (AsyncStorage);
//             tokens/secrets -> secureGet/secureSet/secureRemove (Keychain).
// Values are auto JSON-serialized (string|number|boolean|null) in this implementation — never JSON.stringify/parse yourself.
// Helpers NEVER throw: a miss returns `fallback`, a failed write returns `false` (failures are SILENT).
// 
// Use async/await for all storage operations.
//
// AUTH TOKENS: use ONE namespace (secure*) + ONE shared key constant, and read/write it the SAME
// way on both sides — the login/AuthContext (write) and the API client/interceptor (read). A
// mismatched method or key silently returns the fallback, surfacing as a logged-out state or 401/403
// with no error in the logs.
//
// KEY NAMES: expo-secure-store only accepts [A-Za-z0-9._-]. A key like "auth:token" or "user/session"
// makes every secureSet throw on a phone; the helper swallows it, so the token is never saved and
// login/sign-up silently loop. The web preview does not catch this (no Keychain there).
// Use plain names: "auth_token", "refresh_token".

import AsyncStorage from "@react-native-async-storage/async-storage";
import * as SecureStore from "expo-secure-store";

import { AssertNoExtras, StorageBase, StorageItemValue } from "./storage-base";

export class Storage extends StorageBase {
  // General KV — backed by AsyncStorage.
  // `fallback` is required and returned on any miss/parse error — a missing key looks identical to a stored `null`.
  async getItem<Fallback extends StorageItemValue>(
    key: string,
    fallback: Fallback,
  ): Promise<Fallback | null> {
    try {
      const raw = await AsyncStorage.getItem(key);
      return this.retrieve(raw, fallback);
    } catch (e) {
      this.warn("getItem", key, e);
      return fallback;
    }
  }

  async setItem<Value extends StorageItemValue>(
    key: string,
    value: Value,
  ): Promise<boolean> {
    try {
      await AsyncStorage.setItem(key, JSON.stringify(value));
      return true;
    } catch (e) {
      this.warn("setItem", key, e);
      return false;
    }
  }

  async removeItem(key: string): Promise<boolean> {
    try {
      await AsyncStorage.removeItem(key);
      return true;
    } catch (e) {
      this.warn("removeItem", key, e);
      return false;
    }
  }

  // Sensitive values — Keychain (iOS) / EncryptedSharedPreferences (Android).
  // Use these (not getItem) for auth tokens; whatever writes with secureSet must read with secureGet under the same key.
  // Key must match /^[A-Za-z0-9._-]+$/ — no ":", "/" or spaces — or SecureStore throws and nothing is stored.
  async secureGet<Fallback extends StorageItemValue>(
    key: string,
    fallback: Fallback,
  ): Promise<Fallback | null> {
    try {
      const raw = await SecureStore.getItemAsync(key);
      return this.retrieve(raw, fallback);
    } catch (e) {
      this.warn("secureGet", key, e);
      return fallback;
    }
  }

  async secureSet<Value extends StorageItemValue>(
    key: string,
    value: Value,
  ): Promise<boolean> {
    try {
      await SecureStore.setItemAsync(key, JSON.stringify(value));
      return true;
    } catch (e) {
      this.warn("secureSet", key, e);
      return false;
    }
  }

  async secureRemove(key: string): Promise<boolean> {
    try {
      await SecureStore.deleteItemAsync(key);
      return true;
    } catch (e) {
      this.warn("secureRemove", key, e);
      return false;
    }
  }
}

// The shared singleton — import THIS (`import { storage } from "@/src/utils/storage"`). Do not `new Storage()`.

// eslint-disable-next-line @typescript-eslint/no-unused-vars -- intentional compile-time-only assertion
type _NoExtras = AssertNoExtras<Exclude<keyof Storage, keyof StorageBase>>;

export type ToastData = { message: string; type: "success" | "error" };

export function useToast() {
  const [toast, setToast] = useState<ToastData | null>(null);
  const timer = useRef<ReturnType<typeof setTimeout> | null>(null);
  const show = (message: string, type: "success" | "error" = "success") => {
    setToast({ message, type });
    if (timer.current) clearTimeout(timer.current);
    timer.current = setTimeout(() => setToast(null), 2600);
  };
  useEffect(
    () => () => {
      if (timer.current) clearTimeout(timer.current);
    },
    [],
  );
  return { toast, show };
}

export function ToastItem({ toast }: { toast: ToastData | null }) {
  const styles = useStyles();
  const { colors } = useTheme();
  const insets = useSafeAreaInsets();
  if (!toast) return null;
  return (
    <View
      pointerEvents="none"
      style={[
        styles.toast,
        { top: insets.top + 12, backgroundColor: toast.type === "success" ? colors.success : colors.error },
      ]}
    >
      <Text style={styles.toastText}>{toast.message}</Text>
    </View>
  );
}

// Option-specific policy form modal (Motor / Health / Supreme Healthcare /
// Add Other Policy / Non-USGIC) — back arrow title, red underline, validation
// and submission to the dummy backend API.

import { useEffect, useState } from "react";
import { Pressable, Text, View } from "react-native";
import { useMutation, useQuery, useQueryClient } from "@tanstack/react-query";
import { KeyboardAwareScrollView } from "react-native-keyboard-controller";
import Ionicons from "@react-native-vector-icons/ionicons";

import { api } from "@/src/utils/api";
import type { PolicyCategory } from "@/src/utils/api";
import { Field, GradientButton, ModalShell, SelectField } from "@/src/components/ui";
import { makeStyles, useTheme } from "@/src/theme";

const DATE_RE = /^\d{2}\/\d{2}\/\d{4}$/;

const CATEGORY_TITLE: Record<PolicyCategory, string> = {
  motor: "Add Motor Policy",
  health: "Add Health Policy",
  supreme_healthcare: "Add Supreme Healthcare",
  other: "Add Other Policy",
  non_usgic: "Non-USGIC",
};

type FormValues = {
  policy_no: string;
  engine_or_chassis: string;
  dob: string;
  insurer_name: string;
  plan: string;
  policy_start_date: string;
  policy_end_date: string;
};

const EMPTY: FormValues = {
  policy_no: "",
  engine_or_chassis: "",
  dob: "",
  insurer_name: "",
  plan: "",
  policy_start_date: "",
  policy_end_date: "",
};

function validate(category: PolicyCategory, v: FormValues): Partial<Record<keyof FormValues, string>> {
  const errors: Partial<Record<keyof FormValues, string>> = {};
  const req = (key: keyof FormValues) => {
    if (!v[key].trim()) errors[key] = "This field is required";
  };
  const date = (key: keyof FormValues) => {
    if (v[key].trim() && !DATE_RE.test(v[key].trim())) errors[key] = "Use dd/mm/yyyy format";
  };
  if (category === "motor") {
    req("policy_no");
    req("engine_or_chassis");
  } else if (category === "health" || category === "supreme_healthcare") {
    req("policy_no");
    req("dob");
    date("dob");
  } else if (category === "other") {
    req("insurer_name");
    req("policy_no");
  } else if (category === "non_usgic") {
    req("insurer_name");
    req("plan");
    req("policy_no");
    req("policy_start_date");
    req("policy_end_date");
    date("policy_start_date");
    date("policy_end_date");
  }
  return errors;
}

export function PolicyFormModal(props: {
  visible: boolean;
  category: PolicyCategory | null;
  onClose: () => void;
  onSuccess: () => void;
}) {
  const styles = useStyles();
  const { colors } = useTheme();
  const queryClient = useQueryClient();
  const { category, visible } = props;

  const [values, setValues] = useState<FormValues>(EMPTY);
  const [errors, setErrors] = useState<Partial<Record<keyof FormValues, string>>>({});
  const [formError, setFormError] = useState<string | null>(null);
  const [agree, setAgree] = useState(false);

  useEffect(() => {
    if (visible) {
      setValues(EMPTY);
      setErrors({});
      setFormError(null);
      setAgree(false);
    }
  }, [visible, category]);

  const insurersQuery = useQuery({
    queryKey: ["insurers"],
    queryFn: api.getInsurers,
    enabled: visible && category === "non_usgic",
  });
  const insurers = insurersQuery.data ?? [];
  const selectedInsurer = insurers.find((i) => i.name === values.insurer_name);

  const mutation = useMutation({ mutationFn: api.addPolicy });

  const set = (key: keyof FormValues) => (text: string) =>
    setValues((v) => ({ ...v, [key]: text }));

  const submit = async () => {
    if (!category) return;
    const nextErrors = validate(category, values);
    setErrors(nextErrors);
    setFormError(null);
    if (category === "non_usgic" && !agree) {
      setFormError("Please agree to the Terms & Conditions");
      return;
    }
    if (Object.keys(nextErrors).length > 0) return;
    try {
      await mutation.mutateAsync({
        category,
        policy_no: values.policy_no,
        engine_or_chassis: values.engine_or_chassis,
        dob: values.dob,
        insurer_name: values.insurer_name,
        plan: values.plan,
        policy_start_date: values.policy_start_date,
        policy_end_date: values.policy_end_date,
      });
      await queryClient.invalidateQueries({ queryKey: ["policies"] });
      props.onSuccess();
    } catch (e) {
      setFormError(e instanceof Error ? e.message : "Something went wrong. Please try again.");
    }
  };

  return (
    <ModalShell
      visible={visible}
      onClose={props.onClose}
      onBack={props.onClose}
      title={category ? CATEGORY_TITLE[category] : ""}
      leading="back"
    >
      <KeyboardAwareScrollView
        keyboardShouldPersistTaps="handled"
        bottomOffset={24}
        showsVerticalScrollIndicator={false}
        contentContainerStyle={styles.scrollContent}
      >
        {!!formError && (
          <Text style={styles.formError} testID="policy-form-error">
            {formError}
          </Text>
        )}

        {category === "motor" && (
          <>
            <Field
              testID="policy-number-input"
              placeholder="Policy Number (xxxx/xxxxxxxx/xx/xxx) *"
              value={values.policy_no}
              onChangeText={set("policy_no")}
              error={errors.policy_no}
            />
            <Field
              testID="engine-chassis-input"
              placeholder="Engine No./Chassis No. *"
              value={values.engine_or_chassis}
              onChangeText={set("engine_or_chassis")}
              error={errors.engine_or_chassis}
              autoCapitalize="characters"
            />
          </>
        )}

        {(category === "health" || category === "supreme_healthcare") && (
          <>
            <Field
              testID="policy-number-input"
              placeholder="Policy Number (xxxx/xxxxxxxx/xx/xxx) *"
              value={values.policy_no}
              onChangeText={set("policy_no")}
              error={errors.policy_no}
            />
            <Field
              testID="dob-input"
              placeholder="DOB (dd/mm/yyyy) *"
              value={values.dob}
              onChangeText={set("dob")}
              error={errors.dob}
            />
          </>
        )}

        {category === "other" && (
          <>
            <Field
              testID="insurer-name-input"
              placeholder="Insurer Name *"
              value={values.insurer_name}
              onChangeText={set("insurer_name")}
              error={errors.insurer_name}
            />
            <Field
              testID="policy-number-input"
              placeholder="Policy Number *"
              value={values.policy_no}
              onChangeText={set("policy_no")}
              error={errors.policy_no}
            />
          </>
        )}

        {category === "non_usgic" && (
          <>
            <SelectField
              testID="insurer-select"
              placeholder="Select Insurer name *"
              value={values.insurer_name}
              options={insurers.map((i) => i.name)}
              onSelect={(name) => setValues((v) => ({ ...v, insurer_name: name, plan: "" }))}
              error={errors.insurer_name}
            />
            <SelectField
              testID="plan-select"
              placeholder="Select Plan *"
              value={values.plan}
              options={selectedInsurer?.plans ?? []}
              onSelect={(plan) => setValues((v) => ({ ...v, plan }))}
              error={errors.plan}
            />
            <Field
              testID="policy-number-input"
              placeholder="Policy No. *"
              value={values.policy_no}
              onChangeText={set("policy_no")}
              error={errors.policy_no}
            />
            <Field
              testID="start-date-input"
              placeholder="Policy Start Date (dd/mm/yyyy) *"
              value={values.policy_start_date}
              onChangeText={set("policy_start_date")}
              error={errors.policy_start_date}
            />
            <Field
              testID="end-date-input"
              placeholder="Policy End Date (dd/mm/yyyy) *"
              value={values.policy_end_date}
              onChangeText={set("policy_end_date")}
              error={errors.policy_end_date}
            />
            <Pressable
              testID="terms-checkbox"
              onPress={() => setAgree((a) => !a)}
              style={styles.checkboxRow}
            >
              <View
                style={[
                  styles.checkbox,
                  agree && { backgroundColor: colors.brandPrimary, borderColor: colors.brandPrimary },
                ]}
              >
                {agree && <Ionicons name="checkmark" size={14} color={colors.onBrandPrimary} />}
              </View>
              <Text style={styles.checkboxText}>
                I agree to <Text style={styles.termsLink}>Terms & Conditions</Text>
              </Text>
            </Pressable>
          </>
        )}

        <View style={styles.submitWrap}>
          <GradientButton
            label="Submit"
            loading={mutation.isPending}
            onPress={submit}
            testID="submit-policy-button"
          />
        </View>
      </KeyboardAwareScrollView>
    </ModalShell>
  );
}
